cmake_minimum_required(VERSION 3.24)
project(face_library LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs imgproc dnn objdetect)
include(FetchContent)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
if(MBEDTLS_ROOT)
  add_subdirectory(${MBEDTLS_ROOT} ${CMAKE_BINARY_DIR}/mbedtls EXCLUDE_FROM_ALL)
else()
  FetchContent_Declare(mbedtls URL https://github.com/Mbed-TLS/mbedtls/releases/download/mbedtls-3.6.4/mbedtls-3.6.4.tar.bz2 URL_HASH SHA256=ec35b18a6c593cf98c3e30db8b98ff93e8940a8c4e690e66b41dfc011d678110)
  FetchContent_MakeAvailable(mbedtls)
endif()
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/../../frontend/public/index.html PAGE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../../frontend/public/index.html ${CMAKE_CURRENT_SOURCE_DIR}/../../frontend/public/app.js ${CMAKE_CURRENT_SOURCE_DIR}/../../frontend/public/camera.js ${CMAKE_CURRENT_SOURCE_DIR}/../../frontend/public/style.css ${CMAKE_CURRENT_SOURCE_DIR}/../../frontend/public/dm-sans.woff2)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/../../frontend/public/app.js APPJS)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/../../frontend/public/camera.js CAMERAJS)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/../../frontend/public/dm-sans.woff2 DMSANS_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," DMSANS_BYTES "${DMSANS_HEX}")
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/../../frontend/public/style.css STYLE)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/cacert.pem CACERT)
configure_file(src/assets.h.in generated/assets.h @ONLY)
add_library(sqlite STATIC ${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/sqlite3.c)
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0 SQLITE_DEFAULT_MEMSTATUS=0 SQLITE_OMIT_LOAD_EXTENSION)
add_executable(face-library src/main.cpp)
target_include_directories(face-library PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor ${CMAKE_BINARY_DIR}/generated)
target_compile_definitions(face-library PRIVATE CPPHTTPLIB_THREAD_POOL_COUNT=3)
target_compile_options(face-library PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(face-library PRIVATE ${OpenCV_LIBS} sqlite mbedtls mbedx509 mbedcrypto)
if(UNIX AND NOT APPLE)
  target_link_libraries(face-library PRIVATE pthread dl)
endif()
option(FACE_LIBRARY_TESTS "Build the standalone HNSW index test" OFF)
if(FACE_LIBRARY_TESTS)
  enable_testing()
  add_executable(index-test test/index.cpp)
  target_include_directories(index-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor ${CMAKE_BINARY_DIR}/generated)
  target_link_libraries(index-test PRIVATE ${OpenCV_LIBS} sqlite mbedtls mbedx509 mbedcrypto)
  add_test(NAME index-scale COMMAND index-test)
  add_executable(make-fixtures test/make-fixtures.cpp)
  target_link_libraries(make-fixtures PRIVATE ${OpenCV_LIBS})
  add_executable(orientation-test test/orientation.cpp)
  target_link_libraries(orientation-test PRIVATE ${OpenCV_LIBS})
endif()
